cmake_minimum_required(VERSION 3.16...3.28)
project(Project1 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

add_subdirectory(engine)
add_subdirectory(galaga)

add_executable(${PROJECT_NAME} ${GALAGA})
target_compile_features(${PROJECT_NAME} PRIVATE cxx_std_20)
target_include_directories(${PROJECT_NAME} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/engine/)
target_include_directories(${PROJECT_NAME} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/galaga/)
target_link_libraries(${PROJECT_NAME} Engine)

include(CTest)
if(BUILD_TESTING)
    add_executable(CoreTests tests/CoreTests.cpp galaga/Bullet.cpp galaga/Enemy.cpp galaga/Player.cpp galaga/Galaga.cpp galaga/Stars.cpp)
    target_include_directories(CoreTests PRIVATE galaga)
    target_link_libraries(CoreTests PRIVATE Engine)
    add_test(NAME core COMMAND CoreTests)
    option(ENABLE_WINDOW_TESTS "Build Linux window lifecycle test (requires an X display)" OFF)
    if(ENABLE_WINDOW_TESTS)
        find_package(X11 REQUIRED)
        add_executable(EngineLoopTests tests/EngineLoopTests.cpp)
        target_link_libraries(EngineLoopTests PRIVATE Engine X11::X11)
        add_test(NAME engine_loop COMMAND EngineLoopTests)
        set_tests_properties(engine_loop PROPERTIES TIMEOUT 15)
        add_executable(GameplayTests tests/GameplayTests.cpp galaga/Bullet.cpp galaga/Enemy.cpp galaga/Player.cpp galaga/Galaga.cpp galaga/Stars.cpp)
        target_include_directories(GameplayTests PRIVATE galaga)
        target_link_libraries(GameplayTests PRIVATE Engine)
        add_test(NAME gameplay COMMAND GameplayTests)
        set_tests_properties(gameplay PROPERTIES TIMEOUT 30)
    endif()
endif()
